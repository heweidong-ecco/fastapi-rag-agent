# 批④-B · MCP 长驻会话（actor 模式）· 施工单

> **For agentic workers:** 按 task 顺序做，每个 task 自己带测试周期。步骤用 `- [ ]` 勾。

**Goal:** 把 `/agent/mcp_chat` 那条链的「**每次工具调用起一次 MCP 子进程**（实测 ~2s）」
换成「**一个长驻会话**（实测 4–7 ms）」—— 做法 = **一个 holder task 独占整个 `async with`**，
别人只往队列投请求。

**Architecture:** `DEC-104` §2.1 的目标 + 本轮调研（`docs/说明/mcp长驻会话-调研-20261008.md`）的实测结论。
🔴 **核心不是"共享一个对象"，是"别让别人碰那个 context manager"** —— anyio 只要求
**进入与退出在同一个 task**；被谁 await 过它不管。

**Tech Stack:** Python 3.10 · `mcp` 2.3.0 · anyio

**Spec:** 无前置 spec ⇒ 本次的**实测依据**就是那份调研文档（§二–§四）。

```python
"""长驻会话（批④-B）的口径守卫。⚠️ 全离线：monkeypatch 掉真的 stdio_client，⛔ 不起子进程。"""
import asyncio
import pytest


def _install_fakes(monkeypatch, m, events):
    """装一套假的 stdio_client / ClientSession，并**记下事件顺序**。"""
    import contextlib

    class _FakeSession:
        def __init__(self, r, w): pass
        async def __aenter__(self): events.append("session:enter"); return self
        async def __aexit__(self, *a): events.append("session:exit")
        async def initialize(self): events.append("initialize")
        async def list_tools(self): events.append("list_tools"); return "TOOLS"
        async def call_tool(self, name, args):
            events.append(f"call_tool:{name}"); return _R(args)

    class _R:
        def __init__(self, args): self.content = [type("C", (), {"text": f"got:{args}"})()]

    @contextlib.asynccontextmanager
    async def _fake_stdio(params):
        events.append("stdio:enter")
        try:
            yield (None, None)
        finally:
            events.append("stdio:exit")

    monkeypatch.setattr(m, "stdio_client", _fake_stdio)
    monkeypatch.setattr(m, "ClientSession", _FakeSession)


def test_initialize_runs_once_across_many_calls(monkeypatch):
    """🔴 **本批的全部收益就在这一条**：会话开一次、`initialize` 只跑一次。

    ⚠️ 反证：把 actor 换回"每次调用自开自关" ⇒ `initialize` 会变成 3 次 ⇒ 本用例红。
    """
    import agent_graph_advanced as m
    events = []
    _install_fakes(monkeypatch, m, events)

    async def _run():
        for i in range(3):
            await m.call_mcp_tool("calculator", {"expression": str(i)})
        await m.aclose_mcp_session()

    asyncio.run(_run())

    assert events.count("initialize") == 1, f"会话被重复初始化了：{events}"
    assert events.count("stdio:enter") == 1, f"子进程被重复起了：{events}"
    assert events.count("call_tool:calculator") == 3
    # ⚠️ 退出只该有一次，且在最后
    assert events[-1] == "stdio:exit", f"退出时机不对：{events}"


def test_session_is_rebuilt_for_a_new_event_loop(monkeypatch):
    """🔴 **两次 `asyncio.run` 用两个 loop**（测试与本仓现有很多调用点都是这样）
    ⇒ actor 必须**认 loop**，⛔ 不能把上一个 loop 的死会话拿去用。

    ⚠️ 反证：去掉按 loop 认领的逻辑 ⇒ 第二次会拿到一个「属于已关闭 loop 的队列」⇒ 本用例红。
    """
    import agent_graph_advanced as m
    events = []
    _install_fakes(monkeypatch, m, events)

    async def _one():
        return await m.call_mcp_tool("calculator", {"expression": "1"})

    assert asyncio.run(_one()) == "got:{'expression': '1'}"
    assert asyncio.run(_one()) == "got:{'expression': '1'}"
    # 两个 loop ⇒ 两套会话 ⇒ initialize 两次
    assert events.count("initialize") == 2, f"第二个 loop 复用了旧会话？{events}"


def test_actor_serdes_concurrent_calls(monkeypatch):
    """并发调用**串行**过同一个会话（MCP 单会话是请求-响应）—— ⛔ 不许交叉。"""
    import agent_graph_advanced as m
    events = []
    _install_fakes(monkeypatch, m, events)

    async def _run():
        rs = await asyncio.gather(*[m.call_mcp_tool("calculator", {"expression": str(i)}) for i in range(5)])
        await m.aclose_mcp_session()
        return rs

    rs = asyncio.run(_run())
    assert len(rs) == 5
    assert events.count("initialize") == 1
```

```python
# ==================== 长驻会话（批④-B · 2026-10-08）====================
# 🔴 **为什么是 actor，⛔ 不是会话池**（2026-09-20 栽过的那次就在上面那段 docstring 里记着）：
#    `stdio_client` 基于 anyio，其 cancel scope 要求「**进入与退出在同一个 task**」。
#    池化的形状是「启动 task 建 · 请求 task 用 · 归还 task 关」⇒ **3 个 task** ⇒ 炸。
#    actor 的形状是「**一个 holder task 独占整个 async with**」⇒ **1 个 task** ⇒ ⛔ 不炸。
#    📌 **可复用的判据**：anyio 只管「进入与退出在不在同一个 task」，**不管中间被谁 await 过**。
#    📄 实测（含三轮数据）⇒ `docs/说明/mcp长驻会话-调研-20261008.md` §四
class _McpSessionActor:
    """持有唯一一个长驻 MCP 会话；调用方投请求、等 future，⛔ 不碰 session。"""

    def __init__(self):
        self._loop = asyncio.get_running_loop()
        self._tx: asyncio.Queue = asyncio.Queue()
        self._ready: asyncio.Future = self._loop.create_future()
        self._task: asyncio.Task | None = None
        self._dead_reason: str | None = None
        self._restarts = 0

    @property
    def loop(self):
        return self._loop

    def is_alive(self) -> bool:
        return self._task is not None and not self._task.done()

    async def start(self) -> None:
        if self._task is not None:
            return
        self._task = asyncio.create_task(self._hold(), name="mcp-session-holder")
        await self._ready          # 等 initialize 完成 —— ⛔ 别让第一个请求自己撞

    async def _hold(self) -> None:
        try:
            async with mcp_session() as session:
                if not self._ready.done():
                    self._ready.set_result(True)
                while True:
                    item = await self._tx.get()
                    if item is None:
                        break
                    name, args, fut = item
                    if fut.done():
                        continue        # 调用方已经不要了（超时/取消）
                    try:
                        if name == "__list_tools__":
                            fut.set_result(await session.list_tools())
                        else:
                            fut.set_result(await session.call_tool(name, args))
                    except Exception as exc:          # noqa: BLE001
                        # ⚠️ 一个请求失败**不该**弄死整个会话（那是现状都没有的脆弱性）
                        fut.set_exception(exc)
        except Exception as exc:                       # noqa: BLE001
            self._dead_reason = f"{type(exc).__name__}: {exc}"
        finally:
            if not self._ready.done():
                self._ready.set_exception(RuntimeError(f"MCP 会话起不来：{self._dead_reason}"))

    async def _request(self, name, args=None):
        if not self.is_alive():
            # 🔴 会话没了（崩了 / 被关了）⇒ **重建**，⛔ 别让调用方拿到一个死队列
            self._restarts += 1
            if self._restarts > 5:
                raise RuntimeError(f"MCP 会话反复起不来（{self._restarts} 次）：{self._dead_reason}")
            self._tx = asyncio.Queue()
            self._ready = self._loop.create_future()
            self._task = None
            await self.start()
        fut = self._loop.create_future()
        await self._tx.put((name, args, fut))
        return await fut

    async def call_tool(self, name: str, args: dict) -> str:
        result = await self._request(name, args)
        if result.content:
            return result.content[0].text
        return "工具返回了空结果"

    async def list_tools(self):
        return await self._request("__list_tools__")

    async def stop(self) -> None:
        if self._task is None:
            return
        await self._tx.put(None)
        try:
            await asyncio.wait_for(self._task, timeout=5)
        except (asyncio.TimeoutError, asyncio.CancelledError):
            self._task.cancel()
        self._task = None
```

**② 模块级：按 loop 认领**

```python
_ACTOR: _McpSessionActor | None = None

async def _get_actor() -> _McpSessionActor:
    """拿（必要时建）当前 loop 的那个 actor。

    🔴 **按 loop 认领**：`asyncio.run` 每次都开一个新 loop，而 actor 的队列/future
       **绑在创建它的那个 loop 上** ⇒ ⛔ 不能跨 loop 复用（本仓很多调用点就是 `asyncio.run`）。
    """
    global _ACTOR
    loop = asyncio.get_running_loop()
    if _ACTOR is None or _ACTOR.loop is not loop:
        _ACTOR = _McpSessionActor()
    if not _ACTOR.is_alive():
        await _ACTOR.start()
    return _ACTOR

async def aclose_mcp_session() -> None:
    """显式关掉长驻会话（`main.py` 的 shutdown 会调它）。"""
    global _ACTOR
    if _ACTOR is not None:
        await _ACTOR.stop()
        _ACTOR = None
```

**③ 换掉两个调用点**

```python
async def call_mcp_tool(tool_name: str, arguments: dict) -> str:
    """通过 MCP Client 调用工具。

    🔴 **2026-10-08（批④-B）改：走【长驻会话】** —— 改前每次调用自开自关（实测 ~2s/次，
    成本几乎全在起子进程 + 重新 import langchain）。现在会话由 `_McpSessionActor` 的

    """
    actor = await _get_actor()
    return await actor.call_tool(tool_name, arguments)
```

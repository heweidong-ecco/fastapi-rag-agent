```js
/** 剥掉注释 —— 只剥**三种**，取【保守】策略。
 *
 * 🔴 靶子是【行为】（前端不许自己认拒答），⛔ 不是「字节里出现过这串字」。
 *    `N14` 的实际事故：那句拒答语被写进 `chat.html` 的**注释**里当例证 ⇒ 守卫红。
 *    ⇒ 谁再写注释解释这条守卫，就再红一次。
 *
 * ⛔ **有意不剥行尾 `//` 注释**：剥它要区分 `https://` 里的 `//`（需要一个真词法器）；
 *    而「剥错」的方向是**守卫变瞎**（假绿）—— 比多报一次糟得多。
 *    `N14` 的事故是**一整块注释**，本剥法覆盖得到。
 */
function stripComments(text) {
  return String(text)
    .replace(/<!--[\s\S]*?-->/g, '')      // HTML 注释
    .replace(/\/\*[\s\S]*?\*\//g, '')     // JS 块注释
    .replace(/^[ \t]*\/\/.*$/gm, '');     // JS【整行】注释
}

/** 纯逻辑：给定一组 `{name, text}`，回「谁在【代码】里认了拒答」。 */
function refusalOwners(sources) {
  return sources
    .filter((s) => stripComments(s.text).includes(REFUSE))
    .map((s) => s.name);
}

test('stripComments：三种注释里的那串【都不算】', () => {
  const s = stripComments(
    '<p>ok</p><!-- 这里写了 无法回答 当例证 -->\n' +
    '/* 块注释里的 无法回答 */\n' +
    '   // 整行注释里的 无法回答\n' +
    'const keep = 1;\n',
  );
  assert.ok(!s.includes(REFUSE), `注释没被剥干净：${JSON.stringify(s)}`);
  assert.ok(s.includes('const keep = 1;'), '剥过头了 —— 把代码也吃掉了');
});

test('🔴 反证：代码里的那串【要算】（防「剥多了 ⇒ 守卫变瞎」）', () => {
  assert.deepStrictEqual(
    refusalOwners([
      { name: 'a.js', text: 'if (t.includes("无法"+"回答")) {}' },
      { name: 'b.html', text: '<!-- 无法回答 -->' },
    ]),
    ['a.js'],
  );
});
```

```python
#!/usr/bin/env python3
"""每个 CI job 必须带 `timeout-minutes` —— 缺了就报。

## 为什么（`§七·2` · `DEC-081` §四）

`ci.yml` 现两个 job 都配了（`syntax` 5 · `offline-tests` 15），但**靠人记得**：
新建第 3 个 job 时忘了配 ⇒ **静默没有上限**。
🔴 **卡住的 job 不消耗失败次数、只消耗时间**，而「卡住」与「跑得很慢」在机器痕迹上
**完全一样**（`docs/复盘/2026-09-17-看到汇总行就以为跑完了.md` §一·附）
⇒ 没有上限时，「卡住」**不是一个显式状态**。

## 为什么是纯文本解析（⛔ 不 `import yaml`）

`PyYAML` 在本机 `venv` 里有（6.0.3），但**不在 `api/requirements.txt`** ⇒ CI 里没有。
⛔ **不为一道门给 demo 镜像加依赖** —— 同「`ruff` 不进 requirements」那条理由。
本脚本**只用标准库**。

## 口径

- job 头 = `jobs:` 之后、**恰好 2 空格缩进**、以 `:` 结尾的行（GitHub Actions 的形状）。
- 该 job 自己的块 = 从它的头到**下一个 job 头**（或文件末）。
- 块内必须有**恰好 4 空格缩进**的 `timeout-minutes:`，且值为**正整数**。

## 用法

    python3 scripts/check_ci_job_timeouts.py [path]   # 默认 .github/workflows/ci.yml

## 退出码

    0  全带（并打印每个 job 的分钟数，让人看得见"尺子有读数"）
    1  有 job 缺 / 值非法 —— **点名**
    2  没能判定（文件不在 / 一个 job 都没解析出来）—— ⛔ 不许当作通过
"""
import re
import sys
from pathlib import Path

DEFAULT = ".github/workflows/ci.yml"

_JOB = re.compile(r"^  ([A-Za-z0-9_-]+):\s*$")
_TIMEOUT = re.compile(r"^    timeout-minutes:\s*(\S+)\s*$")


def jobs_with_timeout(text):
    """⇒ [(job名, 分钟数或 None)]；`jobs:` 之前的内容不参与。"""
    lines = text.splitlines()
    try:
        start = next(i for i, l in enumerate(lines) if l.rstrip() == "jobs:")
    except StopIteration:
        return []
    starts = [(i, m.group(1)) for i, l in enumerate(lines[start + 1:], start + 1)
              if (m := _JOB.match(l))]
    out = []
    for n, (idx, name) in enumerate(starts):
        end = starts[n + 1][0] if n + 1 < len(starts) else len(lines)
        found = None
        for l in lines[idx + 1:end]:
            m = _TIMEOUT.match(l)
            if m:
                found = m.group(1)
                break
        out.append((name, found))
    return out


def main(argv):
    path = Path(argv[1] if len(argv) > 1 else DEFAULT)
    if not path.is_file():
        print(f"🔴 读不到 workflow：{path} —— ⛔ 这【不是】通过", file=sys.stderr)
        return 2

    found = jobs_with_timeout(path.read_text(encoding="utf-8"))
    if not found:
        print(f"🔴 在 {path} 里一个 job 都没解析出来 —— 形状变了？⛔ 这【不是】通过",
              file=sys.stderr)
        return 2

    bad = []
    for name, val in found:
        if val is None:
            bad.append((name, "缺 `timeout-minutes`"))
        elif not val.isdigit() or int(val) <= 0:
            bad.append((name, f"值非法：{val!r}（要正整数）"))

    if bad:
        print(f"🔴 {path}：以下 job 没有可用的超时上限 ——", file=sys.stderr)
        for name, why in bad:
            print(f"   · {name} —— {why}", file=sys.stderr)
        print("\n📌 卡住的 job 只烧时间、不烧失败次数 ⇒ 没有上限时「卡住」不是显式状态。",
              file=sys.stderr)
        print("📄 出处：`docs/待办总表.md` §七·2 · `DEC-081` §四", file=sys.stderr)
        return 1

    print(f"✅ {path}：{len(found)} 个 job 全带超时上限 —— "
          + " · ".join(f"{n}={v}" for n, v in found))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

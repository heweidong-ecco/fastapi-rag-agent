**Spec:** 本次要新建 `docs/specs/query_rewriter.md`；改动面涉及 `docs/specs/api_v1_rag.md` · `answer_with_citations.md` · `hybrid_search.md` · `docs/specs/README.md`

import ast


def _fn(tree, name):
    for n in ast.walk(tree):
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) and n.name == name:
            return n
    return None


def _called_names(fn):
    return {
        getattr(c.func, "id", None) or getattr(c.func, "attr", None)
        for c in ast.walk(fn)
        if isinstance(c, ast.Call)
    }


@pytest.mark.parametrize("func_name", ["unified_search", "rewrite_search_api"])

def _is_rag_llm_call(node):
    """RAG 侧两类调用点：裸 `client.chat.completions.create` 与 `llm*.invoke`。"""
    f = node.func
    if not isinstance(f, ast.Attribute):
        return False
    if f.attr not in ("create", "invoke", "stream", "astream", "ainvoke"):
        return False
    src = ast.unparse(f)
    if "chat.completions" in src:
        return True
    if isinstance(f.value, ast.Name) and f.value.id.startswith("llm"):
        return True
    return isinstance(f.value, ast.Attribute) and f.value.attr.startswith("llm")


def _innermost_llm_functions(tree):
    """每个调用点只归给**最内层**包住它的函数（同 `test_billing_wiring._llm_sites_by_innermost_function`）。"""
    found = {}

    def visit(node, cur):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            cur = node
            found.setdefault(id(node), [node, []])
        if cur is not None and isinstance(node, ast.Call) and _is_rag_llm_call(node):
            found[id(cur)][1].append(node)
        for c in ast.iter_child_nodes(node):
            visit(c, cur)

    visit(tree, None)
    return [v for v in found.values() if v[1]]

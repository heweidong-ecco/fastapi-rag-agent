def drive(rag_mod, llm, **kw):
    """直接调端点函数、把帧**全部拉完**，返回 `(帧列表, llm)`。"""
    from schemas import QuestionRequest

    req = QuestionRequest(question="你好", **kw)

    async def go():
        resp = await rag_mod.stream_search(req, thread_id="t-front", user_name="alice")
        return [f async for f in resp.body_iterator]

    return asyncio.run(go()), llm


def payloads_of(frames):
    """把 SSE 文本帧解成 payload 列表；`[DONE]` 原样保留成字符串。

    ⚠️ **必须 JSON 解码**：本端点 `ensure_ascii=True` ⇒ 中文是 `\\uXXXX`，
       ⛔ 直接做子串匹配恒为假（本仓栽过：假红）。
    """
    out = []
    for f in frames:
        if not f.startswith("data: "):
            continue
        body = f[len("data: "):].strip()
        out.append("[DONE]" if body == "[DONE]" else json.loads(body))
    return out

    ⚠️ **推导型**：从**同一轮**同时取 prompt 与 `sources` 帧，逐条比对 ——
       ⛔ 不是把 `1,2,3` 写死（那样两处各写一遍编号时**照样绿**）。
    """
    llm = _PromptCapturingLLM()
    monkeypatch.setattr(rag_env.rag, "get_llm_stream", lambda: llm)

    frames, _ = drive(rag_env.rag, llm, citations=True)
    payloads = payloads_of(frames)

    sources = next(p["sources"] for p in payloads if isinstance(p, dict) and "sources" in p)
    system_prompt = llm.seen_messages[0]["content"]
    numbered = re.findall(r"\[文档(\d+)来源：([^\]\n]+)\]", system_prompt)

    assert numbered, "prompt 里没有 [文档N来源：…] —— 引用模式没生效，本用例没测到东西"
    assert [s["index"] for s in sources] == [int(n) for n, _ in numbered]
    assert [s["source"] for s in sources] == [src for _, src in numbered]

def compute_cost(model: str, prompt_tokens: int, completion_tokens: int) -> float:
    """一次调用的成本（元）—— **唯一实现**（`DEC-085` 契约 B）。

    ⚠️ 改前同一个算式在**本文件里有两份**（`record_usage` 与 `record_cost`）。
       本刀要用的钱数除了抽取**没有别的复用方式** ⇒ 抽出来三处共用，⛔ 不写第三份。

    return (prompt_tokens / 1000) * pricing["prompt"] + (completion_tokens / 1000) * pricing["completion"]


def usage_summary(llm_obj, response) -> Optional[dict]:
    """把一次响应的用量整理成 `usage` 帧的载荷；**没带用量 ⇒ `None`**。

       （本仓墓碑：`agent_checkpointer.py` 那整段记账从未执行过，而测试全绿）。
    ⚠️ 返回 `None` 时调用方**必须不出帧** —— 「记账了才出帧」是一条判据，不是一个建议。

    prompt_tokens = usage.get("input_tokens", 0)
    completion_tokens = usage.get("output_tokens", 0)

        "model": model,
        "prompt_tokens": prompt_tokens,
        "completion_tokens": completion_tokens,
        "cost_usd": compute_cost(model, prompt_tokens, completion_tokens),
    }

'use strict';
/* 对话页的**纯逻辑**：帧解析 / 引用映射 / 出口判定。⛔ 本文件不碰 DOM、不发请求。
 *
 * 🔴 为什么单独一个文件：硬门 B / C 标着「最容易假完成」，而本仓立场是
 *    「写不出命令的，就是还没核过」。这几条是**能写成命令**的那部分 ⇒ 抽出来用 node --test 钉住。
 *    DOM 渲染 / 滚动 / 按钮态留在 chat.html 里，仍靠 DevTools 手工（见 DEC-085 §六·3）。
 *
 * ⚠️ 形状说明：本仓【没有 package.json】⇒ Node 把 .js 当 CommonJS ⇒ 不能用 import/export。
 *    所以下面是**经典脚本**（浏览器 <script src> 直接用），末尾条件导出给 Node 的 require。
 */

const DONE_SENTINEL = '[DONE]';

/** 把收到的缓冲区切成**完整的** SSE 帧；不完整的尾巴原样留在 rest 里。 */
function parseSseChunk(buffer) {
  const parts = buffer.split('\n\n');
  const rest = parts.pop();                     // 最后一段可能是半截
  const payloads = [];
  for (const block of parts) {
    const line = block.split('\n').find((l) => l.startsWith('data: '));
    if (line === undefined) continue;           // 心跳/注释行（以 ':' 开头）跳过
    const body = line.slice('data: '.length).trim();
    if (body === DONE_SENTINEL) { payloads.push(DONE_SENTINEL); continue; }
    try {
      payloads.push(JSON.parse(body));
    } catch (e) {
      // 🔴 解析不了的帧**跳过**，⛔ 不让它把整条流带崩 —— 服务端加一个未知帧
      //    不该让用户白等一场（前端宁可少显示一帧）。⚠️ 但要打到控制台，别静默。
      console.warn('[sse] 解不出的帧已跳过：', body);
    }
  }
  return { payloads, rest };
}

/** 给一帧 payload 分类。`[DONE]` 是字符串哨兵，⛔ 不是对象。 */
function payloadKind(payload) {
  if (payload === DONE_SENTINEL) return 'done';
  if (payload && typeof payload === 'object') {
    if ('error' in payload) return 'error';
    if ('usage' in payload) return 'usage';
    if ('sources' in payload) return 'sources';
    if ('content' in payload) return 'content';
  }
  return 'unknown';
}

/** 从答案正文里取出引用编号：`[来源:2]` / `[来源:1, 3]` ⇒ [2] / [1, 3]（去重、保序）。 */
function citationIndexes(text) {
  const out = [];
  const re = /\[来源\s*:\s*([0-9]+(?:\s*,\s*[0-9]+)*)\]/g;
  let m;
  while ((m = re.exec(text || '')) !== null) {
    for (const piece of m[1].split(',')) {
      const n = parseInt(piece.trim(), 10);
      if (!Number.isNaN(n) && !out.includes(n)) out.push(n);
    }
  }
  return out;
}

/**
 * 把正文里的引用编号映射成 `sources` 帧里的那几条。
 *
 * 🔴 **必须按 `sources[i].index` 找，⛔ 不是按数组下标** ——
 *    prompt 让模型吐的编号与帧里的 `index` 同源（`DEC-085` 契约 A），
 *    而数组顺序是**另一个**东西；按下标硬猜的话，谁插一项就静默错位。
 * ⚠️ 编号在 `sources` 里找不到 ⇒ **跳过**，⛔ 不抛（模型偶尔会吐一个不存在的号）。
 */
function resolveCitations(text, sources) {
  const byIndex = new Map((sources || []).map((s) => [s.index, s]));
  return citationIndexes(text)
    .map((i) => byIndex.get(i))
    .filter((s) => s !== undefined)
    .map((s) => ({ index: s.index, source: s.source, content: s.content, hit: true }));
}

/**
 * 一次请求怎么收的场。
 *
 * ⚠️ `sawError` **与** `sawDone` 都要看：本仓 RAG 端点**故意**在出错时**不发** `[DONE]`
 *    （`api_v1_rag.py:_on_error` 的 docstring 写明了），所以"没有 [DONE]"本身
 *    既可能是出错、也可能是用户点了停止 —— 要靠 aborted 分流。
 */
function classifyExit({ sawDone, sawError, aborted }) {
  if (aborted) return 'aborted';
  if (sawError) return 'error';
  if (sawDone) return 'done';
  return 'incomplete';
}

/** 费用显示。`null`/`undefined` ⇒ `—`（⛔ 别显示 `$0.000000`，那是在说"这轮不要钱"）。 */
function formatCost(costUsd) {
  if (costUsd === null || costUsd === undefined) return '—';
  return '$' + Number(costUsd).toFixed(6);
}

if (typeof module !== 'undefined' && module.exports) {
  module.exports = {
    DONE_SENTINEL, parseSseChunk, payloadKind, citationIndexes, resolveCitations,
    classifyExit, formatCost,

}

function explainStatus(status) {

async function ask() {
  if (state !== 'idle') return;
  const question = $('q').value.trim();
  if (!question) return;

  const t = loadThreads() || currentThread();
  const threadId = t.current;
  const thread = t.threads.find((x) => x.id === threadId);
  if (thread && thread.title === '新会话') {
    thread.title = question.slice(0, 20);
    saveThreads(t); renderThreads();
  }

  const turn = addTurn(question);
  const aEl = turn.querySelector('.a');
  const metaEl = turn.querySelector('.meta');

  state = '请求中';
  sawDone = false; sawError = false; aborted = false; firstContentAt = null;
  let answer = '', sources = [], usage = null, exit = 'incomplete';
  controller = new AbortController();
  $('send').disabled = true; $('stop').hidden = false;
  sentAt = performance.now();

  try {
    const resp = await fetch('/api/v1/rag/stream_search?thread_id=' + encodeURIComponent(threadId), {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-API-Key': currentKey() },
      // 🔴 三条前端必须遵守的（DEC-085 §3.3）：
      //    ① citations 默认 False ⇒ 不显式传 true 就**根本没有 sources 帧**；
      //    ② conversation_history 由后端按 (user, thread) 取 ⇒ 前端**不传**，
      //       ⛔ 也别传 [] 去"清空"（后端那句 `if not req.conversation_history` 是 falsy 判断，[] 会被无视）。
      body: JSON.stringify({ question, citations: true }),
      signal: controller.signal,
    });

    if (!resp.ok) {
      const { msg, relogin } = explainStatus(resp.status);
      aEl.textContent = msg;
      if (relogin) localStorage.removeItem(KEY_STORE);
      metaEl.textContent = '出错';
      state = relogin ? 'idle' : '出错';
      if (relogin) showLogin();
      return;
    }

    state = '流式中';
    const reader = resp.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';

    // 🔴 **直到连接关闭才收工** —— 见 [DONE] 就 return 会把 `sources` 与 `usage` 整段丢掉
    //    （[DONE] 排在它们**之前**是线上契约，`api_v1_rag.py` 的 `_complete` 明写）。
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const { payloads, rest } = RagSse.parseSseChunk(buffer);
      buffer = rest;
      for (const p of payloads) {
        const kind = RagSse.payloadKind(p);
        if (kind === 'content') {
          if (firstContentAt === null) firstContentAt = performance.now();
          answer += p.content;
          renderAnswer(aEl, answer, sources);       // 有了 sources 才能把引用变成可点的
        } else if (kind === 'sources') {
          sources = p.sources;
          renderAnswer(aEl, answer, sources);
        } else if (kind === 'usage') {
          usage = p.usage;
        } else if (kind === 'error') {
          sawError = true;
          answer += `\n[出错] ${p.error}`;
          renderAnswer(aEl, answer, sources);
        } else if (kind === 'done') {
          sawDone = true;
        }
      }
    }
    exit = RagSse.classifyExit({ sawDone, sawError, aborted });
  } catch (err) {
    if (err.name === 'AbortError') {
      aborted = true;
      exit = 'aborted';
    } else {
      sawError = true;
      exit = 'error';
      aEl.textContent = answer + `\n[网络出错] ${err.message}`;
    }
  } finally {
    state = 'idle';
    controller = null;
    $('send').disabled = false; $('stop').hidden = true;

    // 收场徽标 + 费用行
    if (exit === 'aborted') {
      turn.querySelector('.a').insertAdjacentHTML('beforeend', '<span class="badge">已中断</span>');
    }
    const parts = [];
    if (exit === 'aborted') {
      // 🔴 不是文案选择，是在说事实：usage 只在最后一帧回来，提前关流那帧到不了（DEC-084）
      parts.push('中断未计费');
    } else if (usage) {
      parts.push(`${usage.model}`);
      parts.push(`${usage.prompt_tokens}+${usage.completion_tokens} tok`);
      parts.push(RagSse.formatCost(usage.cost_usd));
    } else {
      parts.push('本轮未计费');
    }
    // 首字延迟**前端自己测**（⛔ 不伪装成服务端指标 —— 本仓有"性能数字要先说清是目标还是实测"的前科）
    if (firstContentAt !== null) parts.push(`首字 ${Math.round(firstContentAt - sentAt)} ms`);
    parts.push({ done: '完成', aborted: '已中断', error: '出错', incomplete: '未收到结束标志' }[exit]);
    metaEl.textContent = parts.join(' · ');
    $('q').value = '';
    $('log').scrollTop = $('log').scrollHeight;
  }
}
